# Cycle boundary v1 normative vectors.
# Heights, windows, block counts, and seconds are unsigned decimal integers.
# The scenario is simulation/cycle_boundary/scenario.py, whose activation
# heights sit at grid boundaries rather than at round numbers, because an
# off-by-one in the grid is invisible at a height in the middle of a window.
#
# Every value is rederived by `tools/cycle-boundary-vectors/verify.py` from the
# founder documents restated by hand in that tool's `expected.py` and from a
# live model run. `expected.py` imports nothing from `simulation/`, so a value
# both sources agree on has been reached from the Founder Constitution and from
# the model independently.

# The grid is shared rather than per-seat. Two seats activated 28,799 heights
# apart in the same window hold the identical span, one block later shifts the
# span by exactly one window, and two seats activated in one block coincide. If
# windows were anchored at each seat's own activation the first of these would
# be false, and a performance reallocation would have no shared window to rank
# uptime in.
schedule.same_window_activations_share_span=true
schedule.next_window_activation_shifts_span=true
schedule.same_block_activations_share_span=true

# A complete 731-window span walked cycle by cycle rather than at its endpoints

# and every one is accepted by the check. The two windows immediately outside
# the span invert to no cycle at all.
